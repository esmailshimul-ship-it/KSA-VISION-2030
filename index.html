`; return; } posts.forEach(post => { const item = document.createElement('div'); item.className = 'message-card'; // চেক করা হচ্ছে ইউজার অলরেডি লাইক দিয়েছে কি না (তাহলে বাটনটি হাইলাইট ও ডিজেবল দেখাবে) const isLiked = localStorage.getItem('liked_' + post.id); const activeStyle = isLiked ?
